===== 示例 10 =====
输入位置：编辑器源码
目录或对象：CalculusBasics 根目录的 Smoke.lean；输入完整文件

import Std

#eval 1 + 1                                             -- 计算表达式的值
#check Nat

def EventuallyNat (P : Nat → Prop) : Prop :=            -- 定义自然数上的最终成立
  ∃ N : Nat, ∀ n : Nat, N ≤ n → P n                     -- 要求存在一个合适的见证

example : EventuallyNat (fun n => 100 ≤ n) := by
  refine ⟨100, ?_⟩                                      -- 选取100，留下其余证明任务
  intro n hn
  exact hn

import Mathlib

open Filter
open scoped Topology

#check Filter
#check Filter.atTop
#check Filter.Tendsto
#check nhds

example {P Q : Nat → Prop}
    (hP : ∀ᶠ n in atTop, P n)
    (hQ : ∀ᶠ n in atTop, Q n) :
    ∀ᶠ n in atTop, P n ∧ Q n := by
  exact hP.and hQ
