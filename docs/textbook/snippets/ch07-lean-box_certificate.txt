theorem box_identity (x r : ℝ) (hx : x ≠ 0) :
    x ^ 2 + 2 * r ^ 3 / x - 3 * r ^ 2 =
      (x - r) ^ 2 * (x + 2 * r) / x := by
  field_simp [hx]	-- 由x≠0清除分母
  <;> ring	-- 核对精确因式分解

theorem box_minimum (x r : ℝ) (hx : 0 < x) (hr : 0 < r) :
    3 * r ^ 2 ≤ x ^ 2 + 2 * r ^ 3 / x := by
  apply sub_nonneg.mp
  rw [box_identity x r (ne_of_gt hx)]
  positivity	-- 平方、正因子与正分母保证非负
